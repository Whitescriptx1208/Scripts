local FS = {}

local function fs_ok()
    return type(writefile) == "function" and type(readfile) == "function"
       and type(isfolder)  == "function" and type(makefolder) == "function"
end

function FS.init()
    if not fs_ok() then return false end
    if not isfolder(FOLDER)  then makefolder(FOLDER)  end
    if not isfolder(SCRIPTS) then makefolder(SCRIPTS) end
    return true
end

function FS.read(path)
    if not fs_ok() or not isfile(path) then return nil end
    local ok, d = pcall(readfile, path); return ok and d or nil
end

function FS.write(path, data)
    if not fs_ok() then return false end
    return (pcall(writefile, path, data))
end

function FS.exists(path)
    if not fs_ok() then return false end
    return isfile(path) or isfolder(path)
end

function FS.load_config()
    if not fs_ok() then return {} end
    local raw = FS.read(CFG); if not raw then return {} end
    local ok, d = pcall(HttpService.JSONDecode, HttpService, raw)
    return ok and d or {}
end

function FS.save_config(t)
    if not fs_ok() then return false end
    local ok, e = pcall(HttpService.JSONEncode, HttpService, t)
    if not ok then return false end
    return FS.write(CFG, e)
end

local U = {}

function U.class(n, p, parent)
    local o = Instance.new(n)
    for k, v in pairs(p or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

function U.corner(r, parent)
    return U.class("UICorner", { CornerRadius = UDim.new(0, r or RADIUS_SM) }, parent)
end

function U.stroke(color, thick, trans, parent)
    return U.class("UIStroke", {
        Color = color or Theme.border,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

function U.grad(parent, c1, c2, rot)
    return U.class("UIGradient", {
        Color = ColorSequence.new(c1, c2),
        Rotation = rot or 0,
    }, parent)
end

function U.pad(t, r, b, l, parent)
    return U.class("UIPadding", {
        PaddingTop    = UDim.new(0, t or 0),
        PaddingRight  = UDim.new(0, r or 0),
        PaddingBottom = UDim.new(0, b or 0),
        PaddingLeft   = UDim.new(0, l or 0),
    }, parent)
end

function U.list(dir, pad, parent, halign, valign)
    return U.class("UIListLayout", {
        FillDirection       = dir or Enum.FillDirection.Vertical,
        Padding             = UDim.new(0, pad or GAP),
        HorizontalAlignment = halign or Enum.HorizontalAlignment.Left,
        VerticalAlignment   = valign or Enum.VerticalAlignment.Top,
        SortOrder           = Enum.SortOrder.LayoutOrder,
    }, parent)
end

function U.tween(obj, time, props, style, dir)
    local t = TweenService:Create(obj, TweenInfo.new(
        time, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out
    ), props)
    t:Play(); return t
end

function U.now() return os.date("%H:%M:%S") end

function U.clip_set(s)
    if type(setclipboard) == "function" then pcall(setclipboard, tostring(s)); return true end
    if type(toclipboard) == "function" then pcall(toclipboard, tostring(s)); return true end
    return false
end

function U.clip_get()
    if type(getclipboard) == "function" then
        local ok, v = pcall(getclipboard); return ok and v or ""
    end
    return ""
end

function U.exec_name()
    if type(identifyexecutor) == "function" then
        local ok, n, v = pcall(identifyexecutor)
        if ok and n then return tostring(n), tostring(v or "") end
    end
    return "Unknown", ""
end

Console = { _subs = {} }

function Console.on(cb) table.insert(Console._subs, cb) end

function Console.push(level, msg)
    local e = { level = level, msg = tostring(msg), time = U.now() }
    table.insert(State.console, e)
    if #State.console > 500 then table.remove(State.console, 1) end
    for _, cb in ipairs(Console._subs) do pcall(cb, e) end
end

function Console.info(m)  Console.push("info",  m) end
function Console.warn(m)  Console.push("warn",  m) end
function Console.error(m) Console.push("error", m) end

do
    local rp, rw = print, warn
    _G.print = function(...)
        local p = {}
        for i = 1, select("#", ...) do p[#p+1] = tostring(select(i, ...)) end
        Console.info(table.concat(p, " ")); pcall(rp, ...)
    end
    _G.warn = function(...)
        local p = {}
        for i = 1, select("#", ...) do p[#p+1] = tostring(select(i, ...)) end
        Console.warn(table.concat(p, " ")); pcall(rw, ...)
    end
end

-- ══════════════════════════════════════════════════════════════════════════
local Exec = {}

function Exec.run(src, label)
    label = label or "script"
    if type(src) ~= "string" or #src == 0 then
        Console.warn("Empty source for " .. label); return false
    end
    if type(loadstring) ~= "function" then
        Console.error("loadstring unavailable"); return false
    end
    local fn = loadstring(src, "@Nexus/" .. label)
    if not fn then Console.error("Compile failed: " .. label); return false end
    local t0 = os.clock()
    local ok, err = pcall(fn)
    local dt = string.format("%.3fs", os.clock() - t0)
    if ok then Console.info(("Ran %s in %s"):format(label, dt))
    else        Console.error(("%s → %s"):format(label, tostring(err))) end
    return ok
end

function Exec.run_file(path)
    local c = FS.read(path)
    if not c then Console.error("Cannot read " .. path); return false end
    return Exec.run(c, path:match("([^/]+)$") or "script")
end

function Exec.run_clip()
    local c = U.clip_get()
    if not c or c == "" then Console.warn("Clipboard empty"); return false end
    return Exec.run(c, "clipboard")
end

local Webhook = {}

function Webhook.send(title, desc, color)
    if not State.config.WebhookEnabled then return end
    if State.config.WebhookURL == "" then return end
    task.spawn(function()
        local body = HttpService:JSONEncode({ embeds = {{
            title = title, description = desc,
            color = color or 0x608CFF,
            footer = { text = "Nexus v" .. VERSION },
            timestamp = DateTime.now():ToIsoDate(),
        }}})
        pcall(function()
            request({
                Url = State.config.WebhookURL, Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body = body,
            })
        end)
    end)
end

local UI = {}

function UI.label(parent, text, size, pos, color, align, bold)
    local l = U.class("TextLabel", {
        Text = text, Size = size, Position = pos or UDim2.new(),
        BackgroundTransparency = 1,
        TextColor3 = color or Theme.text,
        Font = bold and Enum.Font.GothamBold or Enum.Font.Gotham,
        TextSize = State.config.FontSize,
        TextXAlignment = align or Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
    }, parent)
    return l
end

function UI.button(parent, text, size, pos, color, cb, style)
    style = style or "solid"
    local b = U.class("TextButton", {
        Text = text, Size = size, Position = pos or UDim2.new(),
        BackgroundColor3 = color or Theme.accent,
        BackgroundTransparency = (style == "ghost") and 0.75 or 0,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        TextColor3 = Theme.text,
        TextSize = State.config.FontSize,
        AutoButtonColor = false,
    }, parent)
    U.corner(RADIUS_SM, b)
    if style == "outline" then U.stroke(Theme.border_hi, 1, 0, b) end

    local base_color = color or Theme.accent
    local base_bg    = b.BackgroundTransparency

    b.MouseEnter:Connect(function()
        U.tween(b, 0.18, {
            BackgroundColor3 = base_color:Lerp(Color3.new(1,1,1), 0.14),
            BackgroundTransparency = math.max(0, base_bg - 0.15),
        })
    end)
    b.MouseLeave:Connect(function()
        U.tween(b, 0.22, {
            BackgroundColor3 = base_color,
            BackgroundTransparency = base_bg,
        })
    end)
    b.MouseButton1Down:Connect(function()
        U.tween(b, 0.08, { BackgroundTransparency = math.min(1, base_bg + 0.1) })
    end)
    b.MouseButton1Up:Connect(function()
        U.tween(b, 0.14, { BackgroundTransparency = base_bg })
    end)

    if cb then
        b.MouseButton1Click:Connect(function()
            local ok, err = pcall(cb)
            if not ok then Console.error("Button: " .. tostring(err)) end
        end)
    end
    return b
end

function UI.input(parent, placeholder, size, pos, cb)
    local t = U.class("TextBox", {
        Size = size, Position = pos or UDim2.new(),
        BackgroundColor3 = Theme.surface_2,
        BorderSizePixel = 0,
        Font = Enum.Font.Gotham,
        PlaceholderText = placeholder or "",
        PlaceholderColor3 = Theme.text_mute,
        TextColor3 = Theme.text,
        TextSize = State.config.FontSize,
        TextXAlignment = Enum.TextXAlignment.Left,
        ClearTextOnFocus = false,
    }, parent)
    U.corner(RADIUS_SM, t)
    local s = U.stroke(Theme.border, 1, 0, t)
    U.pad(0, 10, 0, 10, t)

    t.Focused:Connect(function()
        U.tween(s, 0.2, { Color = Theme.accent, Transparency = 0.2 })
    end)
    t.FocusLost:Connect(function(enter)
        U.tween(s, 0.2, { Color = Theme.border, Transparency = 0 })
        if cb then pcall(cb, t.Text, enter) end
    end)
    return t
end

local function ensure_toast_host()

    ToastHost = U.class("Frame", {

        Position = UDim2.new(1, -PAD, 1, -PAD),
        Size = UDim2.new(0, math.floor(280 * SCALE) + 40, 1, -40),
        BackgroundTransparency = 1,
    }, gui)
    U.list(Enum.FillDirection.Vertical, 6, ToastHost,
        Enum.HorizontalAlignment.Right, Enum.VerticalAlignment.Bottom)
    return ToastHost

function Notify(title, msg, level)
    level = level or "info"
    local host = ensure_toast_host(); if not host then return end
    local accent = ({
        info = Theme.accent, ok = Theme.green,
        warn = Theme.yellow, error = Theme.red,
    })[level] or Theme.accent

    local h = math.floor(56 * SCALE) + 22
    local f = U.class("Frame", {
        Size = UDim2.new(1, 0, 0, h),
        BackgroundColor3 = Theme.surface,
        BorderSizePixel = 0,
    }, host)
    U.corner(RADIUS_SM, f)
    local st = U.stroke(accent, 1, 0.55, f)
    U.pad(8, 12, 8, 14, f)

    local bar = U.class("Frame", {
        Size = UDim2.new(0, 3, 1, -8),
        Position = UDim2.new(0, 0, 0, 4),
        BackgroundColor3 = accent,
        BorderSizePixel = 0,
    }, f)
    U.corner(2, bar)

    UI.label(f, title, UDim2.new(1, 0, 0, 16), UDim2.new(0, 8, 0, 0), accent, nil, true)
    local m = UI.label(f, msg or "", UDim2.new(1, 0, 1, -18), UDim2.new(0, 8, 0, 18), Theme.text_dim)
    m.TextWrapped = true
    m.TextYAlignment = Enum.TextYAlignment.Top

    f.Position = UDim2.new(1, 60, 0, 0)
    U.tween(f, 0.35, { Position = UDim2.new(0, 0, 0, 0) },
        Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

    task.delay(4.5, function()
        U.tween(f, 0.3, { Position = UDim2.new(1, 60, 0, 0) },
            Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        task.wait(0.35)
        pcall(function()
            U.tween(st, 0.15, { Transparency = 1 })
            f:Destroy()
        end)
    end)
end

-- page switcher with smooth transitions
State._switch_page = function(name)
    if State.active_page == name then return end
    local prev = State.active_page and Pages[State.active_page]

    -- update button states
    for k, b in pairs(sb_buttons) do
        if k == name then
            U.tween(b, 0.2, { TextColor3 = Theme.text })
        else
            U.tween(b, 0.2, { TextColor3 = Theme.text_dim })
        end
    end

    -- slide pill
    local target_y = (sb_buttons[name].LayoutOrder - 1) * (BTN_H + 4)
    U.tween(pill, 0.28, { Position = UDim2.new(0, 0, 0, target_y) },
        Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

    -- fade out old page
    local newp = Pages[name]
    if prev then
        U.tween(prev, 0.12, { BackgroundTransparency = 1 }, Enum.EasingStyle.Quad)
        task.delay(0.1, function() prev.Visible = false end)
    end

    -- fade in new
    task.delay(0.1, function()
        newp.Visible = true
        newp.BackgroundTransparency = 1
        U.tween(newp, 0.2, { BackgroundTransparency = 1 })
    end)

    State.active_page = name
    getgenv().NexusActivePage = name
end

-- ══════════════════════════════════════════════════════════════════════════
--  [20] AUTO-EXEC
-- ══════════════════════════════════════════════════════════════════════════
task.spawn(function()
    task.wait(1)
    if not fs_ok() or type(listfiles) ~= "function" then return end
    local auto_dir = FOLDER .. "/autoexec"
    if not isfolder(auto_dir) then makefolder(auto_dir) end
    local ok, files = pcall(listfiles, auto_dir)
    if not ok or not files then return end
    for _, file in ipairs(files) do
        local path = auto_dir .. "/" .. file
        if isfile(path) then task.spawn(function() Exec.run_file(path) end) end
    end
end)

    end
end)
