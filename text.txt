local FS = {}

local function fs_available()
    return type(writefile) == "function"
        and type(readfile) == "function"
        and type(isfolder) == "function"
        and type(makefolder) == "function"
end

function FS.init()
    if not fs_available() then return false end
    if not isfolder(FOLDER) then makefolder(FOLDER) end
    if not isfolder(SCRIPTS) then makefolder(SCRIPTS) end
    return true
end

function FS.read(path)
    if not fs_available() then return nil end
    if not isfile(path) then return nil end
    local ok, data = pcall(readfile, path)
    return ok and data or nil
end

function FS.write(path, data)
    if not fs_available() then return false end
    local ok = pcall(writefile, path, data)
    return ok
end

function FS.exists(path)
    if not fs_available() then return false end
    return isfile(path) or isfolder(path)
end

function FS.load_config()
    if not fs_available() then return {} end
    local raw = FS.read(CFG)
    if not raw then return {} end
    local ok, decoded = pcall(HttpService.JSONDecode, HttpService, raw)
    return ok and decoded or {}
end

function FS.save_config(tbl)
    if not fs_available() then return false end
    local ok, enc = pcall(HttpService.JSONEncode, HttpService, tbl)
    if not ok then return false end
    return FS.write(CFG, enc)
end

local Util = {}

function Util.class(name, props, parent)
    local obj = Instance.new(name)
    for k, v in pairs(props or {}) do obj[k] = v end
    if parent then obj.Parent = parent end
    return obj
end

function Util.corner(radius, parent)
    return Util.class("UICorner", { CornerRadius = UDim.new(0, radius or 6) }, parent)
end

function Util.stroke(color, thickness, transparency, parent)
    return Util.class("UIStroke", {
        Color = color or Theme.border,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

function Util.padding(t, r, b, l, parent)
    return Util.class("UIPadding", {
        PaddingTop    = UDim.new(0, t or 0),
        PaddingRight  = UDim.new(0, r or 0),
        PaddingBottom = UDim.new(0, b or 0),
        PaddingLeft   = UDim.new(0, l or 0),
    }, parent)
end

function Util.list(fill, pad, parent, halign, valign)
    return Util.class("UIListLayout", {
        FillDirection = fill or Enum.FillDirection.Vertical,
        Padding = UDim.new(0, pad or 6),
        HorizontalAlignment = halign or Enum.HorizontalAlignment.Left,
        VerticalAlignment   = valign or Enum.VerticalAlignment.Top,
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, parent)
end

function Util.tween(obj, time, props, style, dir)
    local t = TweenService:Create(obj, TweenInfo.new(
        time, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out
    ), props)
    t:Play()
    return t
end

function Util.now()
    return os.date("%H:%M:%S")
end

function Util.clipboard_set(s)
    if type(setclipboard) == "function" then pcall(setclipboard, tostring(s)) return true end
    if type(toclipboard) == "function" then pcall(toclipboard, tostring(s)) return true end
    return false
end

function Util.clipboard_get()
    if type(getclipboard) == "function" then
        local ok, v = pcall(getclipboard)
        return ok and v or ""
    end
    return ""
end

function Util.executor_name()
    if type(identifyexecutor) == "function" then
        local ok, name, ver = pcall(identifyexecutor)
        if ok and name then return tostring(name), tostring(ver or "") end
    end
    return "Unknown", ""
end

local Exec = {}

function Exec.run(source, label)
    label = label or "script"
    if type(source) ~= "string" or #source == 0 then
        Console.warn("Empty source for " .. label)
        return false
    end

    local fn
    if type(loadstring) == "function" then
        fn = loadstring(source, "@Nexus/" .. label)
    else
        Console.error("loadstring unavailable on this executor")
        return false
    end

    if not fn then
        Console.error("Compile failed for " .. label)
        return false
    end

    local start = os.clock()
    local ok, err = pcall(fn)
    local elapsed = string.format("%.3fs", os.clock() - start)

    if ok then
        Console.info(string.format("Ran %s in %s", label, elapsed))
    else
        Console.error(string.format("%s → %s", label, tostring(err)))
    end
    return ok
end

function Exec.run_file(path)
    local content = FS.read(path)
    if not content then
        Console.error("Cannot read " .. path)
        return false
    end
    return Exec.run(content, path:match("([^/]+)$") or "script")
end

function Exec.run_clipboard()
    local content = Util.clipboard_get()
    if not content or content == "" then
        Console.warn("Clipboard empty")
        return false
    end
    return Exec.run(content, "clipboard")
end

Console = {
    _events = {},
}

function Console.on(callback)
    table.insert(Console._events, callback)
end

function Console.push(level, msg)
    local entry = {
        level = level,
        msg = tostring(msg),
        time = Util.now(),
    }
    table.insert(State.console, entry)
    if #State.console > 500 then table.remove(State.console, 1) end
    for _, cb in ipairs(Console._events) do
        pcall(cb, entry)
    end
end

function Console.info(m)  Console.push("info", m)  end
function Console.warn(m)  Console.push("warn", m)  end
function Console.error(m) Console.push("error", m) end

-- hook global print / warn / error
do
    local raw_print = print
    local raw_warn  = warn
    local raw_error = error

    _G.print = function(...)
        local parts = {}
        for i = 1, select("#", ...) do
            table.insert(parts, tostring(select(i, ...)))
        end
        Console.info(table.concat(parts, " "))
        pcall(raw_print, ...)
    end

    _G.warn = function(...)
        local parts = {}
        for i = 1, select("#", ...) do
            table.insert(parts, tostring(select(i, ...)))
        end
        Console.warn(table.concat(parts, " "))
        pcall(raw_warn, ...)
    end
end

local Webhook = {}

function Webhook.send(title, desc, color)
    if not State.config.WebhookEnabled then return end
    if not State.config.WebhookURL or State.config.WebhookURL == "" then return end

    task.spawn(function()
        local payload = {
            embeds = {{
                title = title,
                description = desc,
                color = color or 0x5A8CFF,
                footer = { text = "Nexus Executor v" .. VERSION },
                timestamp = DateTime.now():ToIsoDate(),
            }}
        }
        local ok, enc = pcall(HttpService.JSONEncode, HttpService, payload)
        if not ok then return end
        pcall(function()
            request({
                Url = State.config.WebhookURL,
                Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body = enc,
            })
        end)
    end)
end

    return Util.class("TextLabel", {
        Text = text,
        Size = size,
        Position = pos or UDim2.new(),
        BackgroundTransparency = 1,
        TextColor3 = color or Theme.text,
        Font = Enum.Font.Gotham,
        TextSize = State.config.FontSize,
        TextXAlignment = align or Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
    }, parent)
end

function UI.button(parent, text, size, pos, color, callback)
    local b = Util.class("TextButton", {
        Text = text,
        Size = size,
        Position = pos or UDim2.new(),
        BackgroundColor3 = color or Theme.accent,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        TextColor3 = Theme.text,
        TextSize = State.config.FontSize,
        AutoButtonColor = false,
    }, parent)
    Util.corner(6, b)

    local enter = b.MouseEnter:Connect(function()
        Util.tween(b, 0.15, { BackgroundColor3 = color and color:Lerp(Color3.new(1,1,1), 0.15) or Theme.accent:Lerp(Color3.new(1,1,1), 0.15) })
    end)
    local leave = b.MouseLeave:Connect(function()
        Util.tween(b, 0.15, { BackgroundColor3 = color or Theme.accent })
    end)

    b.MouseButton1Click:Connect(function()
        if callback then
            local ok, err = pcall(callback)
            if not ok then Console.error("Button error: " .. tostring(err)) end
        end
    end)
    return b
end

function UI.input(parent, placeholder, size, pos, callback)
    local box = Util.class("TextBox", {
        Size = size,
        Position = pos or UDim2.new(),
        BackgroundColor3 = Theme.surface_2,
        BorderSizePixel = 0,
        Font = Enum.Font.Gotham,
        Text = "",
        PlaceholderText = placeholder or "",
        PlaceholderColor3 = Theme.text_dim,
        TextColor3 = Theme.text,
        TextSize = State.config.FontSize,

        task.wait(0.35)
        pcall(function() frame:Destroy() end)
    end)

-- ── Drag ─────────────────────────────────────────────────────────────────
do
    local dragging, drag_input, start_pos, start_offset
    local drag_bar = nil -- set after topbar created

    local function begin_drag(input)
        dragging = true
        drag_input = input
        start_pos = input.Position
        start_offset = root.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end

    local function update_drag(input)
        if not dragging then return end
        local delta = input.Position - start_pos
        root.Position = UDim2.new(
            start_offset.X.Scale, start_offset.X.Offset + delta.X,
            start_offset.Y.Scale, start_offset.Y.Offset + delta.Y
        )
    end

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            update_drag(input)
        elseif dragging and input.UserInputType == Enum.UserInputType.Touch then
            update_drag(input)
        end
    end)

    -- expose begin_drag for topbar
    State._begin_drag = begin_drag
end

--  [17] AUTO-EXECUTE
-- ══════════════════════════════════════════════════════════════════════════
task.spawn(function()
    task.wait(1)
    if not fs_available() then return end

    local auto_dir = FOLDER .. "/autoexec"
    if not isfolder(auto_dir) then makefolder(auto_dir) end

    if type(listfiles) ~= "function" then return end
    local ok, files = pcall(listfiles, auto_dir)
    if not ok or not files then return end

    for _, file in ipairs(files) do
        local path = auto_dir .. "/" .. file
        if isfile(path) then
            task.spawn(function()
                Exec.run_file(path)
            end)
        end
    end
end)
